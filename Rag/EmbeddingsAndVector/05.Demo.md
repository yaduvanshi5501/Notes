# EmbeddingController.java
```
package com.infy.embeddings.controller;

import java.util.List;

import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

import com.infy.embeddings.model.SearchResult;
import com.infy.embeddings.service.EmbeddingService;

@RestController
@RequestMapping("/embeddings")
public class EmbeddingController {

    private final EmbeddingService embeddingService;

    public EmbeddingController(EmbeddingService embeddingService) {
        this.embeddingService = embeddingService;
    }

    // Generates an embedding vector for the given text
    @GetMapping("/embed")
    public float[] getEmbedding(
            @RequestParam(defaultValue = "CAT") String text) {

        return embeddingService.getEmbedding(text);
    }

    // Converts documents into embeddings and stores them in the Vector Store
    @PostMapping("/store")
    public String createEmbeddings() {

        embeddingService.createEmbeddings();

        return "Embeddings stored successfully!";
    }

    // Semantic similarity search
    @GetMapping("/search")
    public List<SearchResult> searchNearestNeighbors(

            @RequestParam(defaultValue = "How many WFH days are allowed?")
            String query) {

        return embeddingService.searchNearestNeighbors(query)
                .stream()
                .map(SearchResult::from)
                .toList();
    }

    // Semantic search + metadata filtering (Hybrid Search)
    @GetMapping("/search-filtered")
    public List<SearchResult> searchWithMetadataFilter(

            @RequestParam(defaultValue = "health insurance benefits")
            String query,

            @RequestParam(defaultValue = "HR")
            String department,

            @RequestParam(defaultValue = "high")
            String priority,

            @RequestParam(defaultValue = "2")
            int topK) {

        return embeddingService
                .searchWithMetadataFilter(
                        query,
                        department,
                        priority,
                        topK)
                .stream()
                .map(SearchResult::from)
                .toList();
    }

    // Returns only documents above the similarity score
    @GetMapping("/search-threshold")
    public List<SearchResult> searchWithSimilarityThreshold(

            @RequestParam(defaultValue = "employee benefits")
            String query,

            @RequestParam(defaultValue = "0.5")
            double similarityThreshold,

            @RequestParam(defaultValue = "3")
            int topK) {

        return embeddingService
                .searchWithSimilarityThreshold(
                        query,
                        similarityThreshold,
                        topK)
                .stream()
                .map(SearchResult::from)
                .toList();
    }

}
```
# SearchResult.java
```
package com.infy.embeddings.model;

import java.util.Map;

import org.springframework.ai.document.Document;

// DTO returned to the client instead of exposing Spring AI Document
public record SearchResult(

        String content,
        Map<String, Object> metadata,
        Double score) {

    // Convert Spring AI Document -> SearchResult
    public static SearchResult from(Document document) {

        return new SearchResult(
                document.getText(),
                document.getMetadata(),
                document.getScore());
    }

}
```

# application.properties (Azure OpenAI)
```
# Application Name
spring.application.name=embeddings

# Azure OpenAI
spring.ai.azure.openai.api-key=${AZURE_OPENAI_KEY}
spring.ai.azure.openai.endpoint=${AZURE_OPENAI_ENDPOINT}

# Embedding deployment name
spring.ai.azure.openai.embedding.options.model=text-embedding-ada-002

# MariaDB
spring.datasource.url=jdbc:mariadb://localhost:3306/vectordb
spring.datasource.username=${DB_USERNAME}
spring.datasource.password=${DB_PASSWORD}
spring.datasource.driver-class-name=org.mariadb.jdbc.Driver

# Optional when using JPA
spring.jpa.database-platform=org.hibernate.dialect.MariaDBDialect

# Vector Store
spring.ai.vectorstore.mariadb.initialize-schema=true

# Similarity metric
# COSINE -> Recommended for text embeddings
spring.ai.vectorstore.mariadb.distance-type=COSINE

# Must match embedding model dimension
spring.ai.vectorstore.mariadb.dimensions=1536

# true  -> Drops table every startup
# false -> Keeps existing vectors
spring.ai.vectorstore.mariadb.remove-existing-vector-store-table=true
```

# SQL Script

```
DROP DATABASE IF EXISTS vectordb;

CREATE DATABASE vectordb;

USE vectordb;

DROP TABLE IF EXISTS vector_store;

CREATE TABLE vector_store (

    id VARCHAR(255) PRIMARY KEY,

    content TEXT NOT NULL,

    metadata JSON,

    -- Vector dimension must match embedding model
    embedding VECTOR(1536) NOT NULL,

    INDEX idx_content_prefix(content(100))

) ENGINE=InnoDB
DEFAULT CHARSET=utf8mb4
COLLATE=utf8mb4_unicode_ci;
```
